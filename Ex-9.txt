Library Management System – UDP

server.c


#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <time.h>

#define PORT 8080
#define MAX 512

struct Book
{
    int id;
    char name[30];
    int available;
};

struct Book b[3] =
{
    {101, "C_Programming", 1},
    {102, "Python", 1},
    {103, "Java", 1}
};

int main()
{
    int sock, n, i, choice, id;
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    char buffer[MAX], response[MAX];

    sock = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(PORT);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(sock, (struct sockaddr *)&server, sizeof(server));

    printf("Library UDP Server Started...\n");

    while(1)
    {
        n = recvfrom(sock, buffer, MAX-1, 0,
            (struct sockaddr *)&client, &len);

        if(n <= 0)
            continue;

        buffer[n] = '\0';

        int reqid;
        sscanf(buffer, "%d %d %d", &reqid, &choice, &id);

        printf("Request Received: %s\n", buffer);

        if(choice == 1)
        {
            strcpy(response, "Available Books: ");

            for(i = 0; i < 3; i++)
            {
                if(b[i].available)
                {
                    char temp[50];
                    sprintf(temp, "%d-%s ", b[i].id, b[i].name);
                    strcat(response, temp);
                }
            }
        }
        else if(choice == 2 || choice == 3)
        {
            int found = 0;

            for(i = 0; i < 3; i++)
            {
                if(b[i].id == id)
                {
                    found = 1;

                    if(choice == 2 && b[i].available)
                    {
                        b[i].available = 0;
                        sprintf(response, "Book %d Issued Successfully", id);
                    }
                    else if(choice == 3 && !b[i].available)
                    {
                        b[i].available = 1;
                        sprintf(response, "Book %d Returned Successfully", id);
                    }
                    else
                    {
                        strcpy(response, "Book Status Invalid");
                    }

                    break;
                }
            }

            if(!found)
                strcpy(response, "Book Not Found");
        }
        else if(choice == 4)
        {
            int found = 0;

            for(i = 0; i < 3; i++)
            {
                if(b[i].id == id)
                {
                    sprintf(response, "Book: %s | Status: %s",
                        b[i].name,
                        b[i].available ? "Available" : "Issued");

                    found = 1;
                    break;
                }
            }

            if(!found)
                strcpy(response, "Book Not Found");
        }
        else
        {
            strcpy(response, "Invalid Request");
        }

        char logmsg[100];
        sprintf(logmsg, "Request ID %d, Choice %d, Book %d",
                reqid, choice, id);

        FILE *fp = fopen("library_log.txt", "a");

        if(fp != NULL)
        {
            time_t now = time(NULL);
            fprintf(fp, "%s : %s", logmsg, ctime(&now));
            fclose(fp);
        }

        sendto(sock, response, strlen(response), 0,
            (struct sockaddr *)&client, len);

        printf("Response Sent: %s\n", response);
    }

    close(sock);
    return 0;
}

    sock = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(PORT);
    inet_pton(AF_INET, "127.0.0.1", &server.sin_addr);

    struct timeval tv;
    tv.tv_sec = 2;
    tv.tv_usec = 0;

    setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
               &tv, sizeof(tv));

    while(1)
    {
        printf("\n1. View Available Books");
        printf("\n2. Issue Book");
        printf("\n3. Return Book");
        printf("\n4. Search Book");
        printf("\n5. Exit");

        printf("\nEnter Choice: ");
        scanf("%d", &choice);

        id = 0;

        if(choice >= 2 && choice <= 4)
        {
            printf("Enter Book ID: ");
            scanf("%d", &id);
        }

        if(choice == 5)
            break;

        if(choice < 1 || choice > 4)
        {
            printf("Invalid Choice\n");
            continue;
        }

        sprintf(buffer, "%d %d %d", reqid, choice, id);

        for(tries = 1; tries <= 3; tries++)
        {
            sendto(sock, buffer, strlen(buffer), 0,
                (struct sockaddr *)&server, sizeof(server));

            n = recvfrom(sock, buffer, MAX-1, 0, NULL, NULL);

            if(n > 0)
            {
                buffer[n] = '\0';
                printf("\nSERVER: %s\n", buffer);
                break;
            }

            printf("Timeout! Retransmitting Request...\n");
        }

        if(tries > 3)
            printf("Server Not Responding\n");

        reqid++;
    }

    close(sock);
    return 0;
}

2. Real-Time Weather Monitoring System – UDP

server.c


#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8081
#define MAX 512

struct Weather
{
    char location[30];
    float temp, humidity, pressure, wind;
    float st, sh, sp, sw;
    int count;
    int lastseq;
};

struct Weather w[10];
int locations = 0;

int main()
{
    int sock, n, i, found, seq;
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    char buffer[MAX], response[MAX];
    char location[30], type[10];

    float t, h, p, wind;

    sock = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(PORT);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(sock, (struct sockaddr *)&server, sizeof(server));

    printf("Weather UDP Server Started...\n");

    while(1)
    {
        n = recvfrom(sock, buffer, MAX-1, 0,
            (struct sockaddr *)&client, &len);

        if(n <= 0)
            continue;

        buffer[n] = '\0';

        sscanf(buffer, "%9s", type);

        if(strcmp(type, "DATA") == 0)
        {
            sscanf(buffer, "%*s %d %29s %f %f %f %f",
                &seq, location, &t, &h, &p, &wind);

            found = -1;

            for(i = 0; i < locations; i++)
            {
                if(strcmp(w[i].location, location) == 0)
                {
                    found = i;
                    break;
                }
            }

            if(found == -1 && locations < 10)
            {
                found = locations++;
                strcpy(w[found].location, location);
                w[found].lastseq = 0;
            }

            if(found == -1)
            {
                strcpy(response, "ERROR: Location Limit Reached");
            }
            else if(seq <= w[found].lastseq)
            {
                sprintf(response, "ACK %d DUPLICATE", seq);
            }
            else
            {
                if(w[found].lastseq != 0 &&
                   seq > w[found].lastseq + 1)
                {
                    printf("Missing packets: %d to %d\n",
                        w[found].lastseq + 1, seq - 1);
                }

                w[found].lastseq = seq;

                w[found].st += t;
                w[found].sh += h;
                w[found].sp += p;
                w[found].sw += wind;

                w[found].count++;

                sprintf(response, "ACK %d RECEIVED", seq);

                printf("Weather Data Received: %s\n", location);
            }
        }
        else if(strcmp(type, "QUERY") == 0)
        {
            sscanf(buffer, "%*s %29s", location);

            found = -1;

            for(i = 0; i < locations; i++)
            {
                if(strcmp(w[i].location, location) == 0)
                {
                    found = i;
                    break;
                }
            }

            if(found == -1 || w[found].count == 0)
            {
                strcpy(response, "No Weather Data Available");
            }
            else
            {
                sprintf(response,
                    "Location: %s | Avg Temp: %.2f | Avg Humidity: %.2f | Avg Pressure: %.2f | Avg Wind: %.2f",
                    location,
                    w[found].st / w[found].count,
                    w[found].sh / w[found].count,
                    w[found].sp / w[found].count,
                    w[found].sw / w[found].count);
            }
        }
        else
        {
            strcpy(response, "Invalid Request");
        }

        sendto(sock, response, strlen(response), 0,
            (struct sockaddr *)&client, len);

        printf("Response: %s\n", response);
    }

    close(sock);
    return 0;
}

#include <stdio.h>

#include <unistd.h>
#include <arpa/inet.h>
#include <sys/time.h>

#define PORT 8081
#define MAX 512

int main()
{
    int sock, choice, seq = 1, n, tries;
    struct sockaddr_in server;
    char buffer[MAX], location[30];
    float t, h, p, wind;

    sock = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(PORT);
    inet_pton(AF_INET, "127.0.0.1", &server.sin_addr);

    struct timeval tv;
    tv.tv_sec = 2;
    tv.tv_usec = 0;

    setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
               &tv, sizeof(tv));

    while(1)
    {
        printf("\n1. Send Weather Data");
        printf("\n2. View Average Weather");
        printf("\n3. Exit");

        printf("\nEnter Choice: ");
        scanf("%d", &choice);

        if(choice == 1)
        {
            printf("Enter Location: ");
            scanf("%29s", location);

            printf("Enter Temperature: ");
            scanf("%f", &t);

            printf("Enter Humidity: ");
            scanf("%f", &h);

            printf("Enter Pressure: ");
            scanf("%f", &p);

            printf("Enter Wind Speed: ");
            scanf("%f", &wind);

            sprintf(buffer,
                "DATA %d %s %.2f %.2f %.2f %.2f",
                seq, location, t, h, p, wind);
        }
        else if(choice == 2)
        {
            printf("Enter Location: ");
            scanf("%29s", location);

            sprintf(buffer, "QUERY %s", location);
        }
        else if(choice == 3)
        {
            break;
        }
        else
        {
            printf("Invalid Choice\n");
            continue;
        }

        for(tries = 1; tries <= 3; tries++)
        {
            sendto(sock, buffer, strlen(buffer), 0,
                (struct sockaddr *)&server, sizeof(server));

            if(n > 0)
            {
                buffer[n] = '\0';
