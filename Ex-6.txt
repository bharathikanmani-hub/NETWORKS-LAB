Online Railway Reservation System – Iterative TCP

server.c


#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <time.h>

#define PORT 8080
#define MAX 512

struct Train
{
    int id;
    char name[30];
    char from[20];
    char to[20];
    int seats;
    int fare;
};

struct Booking
{
    int id;
    int train;
    char name[30];
    int seats;
    int active;
};

struct Train t[3] =
{
    {101, "ChennaiExpress", "Chennai", "Coimbatore", 10, 250},
    {102, "PandianExpress", "Madurai", "Chennai", 15, 300},
    {103, "CheranExpress", "Coimbatore", "Chennai", 20, 280}
};

struct Booking b[100];

int count = 0;
int booking_id = 1001;

void log_activity(char *msg)
{
    FILE *fp = fopen("railway_log.txt", "a");

    if (fp != NULL)
    {
        time_t now = time(NULL);
        fprintf(fp, "%s : %s", msg, ctime(&now));
        fclose(fp);
    }
}

void send_msg(int sock, char *msg)
{
    char buffer[2048];

    snprintf(buffer, sizeof(buffer), "%s\n", msg);

    send(sock, buffer, strlen(buffer), 0);
}

void receive_msg(int sock, char *msg)
{
    int i = 0;
    char ch;

    while (i < MAX - 1)
    {
        if (recv(sock, &ch, 1, 0) <= 0)
        {
            msg[i] = '\0';
            return;
        }

        if (ch == '\n')
            break;

        msg[i++] = ch;
    }

    msg[i] = '\0';
}

void handle_client(int sock)
{
    char buffer[MAX], response[2048];
    int choice, id, seats, i, found;

    send_msg(sock, "ONLINE RAILWAY RESERVATION SYSTEM");

    while (1)
    {
        receive_msg(sock, buffer);

        if (buffer[0] == '\0')
            break;

        sscanf(buffer, "%d", &choice);

        if (choice == 1)
        {
            strcpy(response, "AVAILABLE TRAINS:\n");

            for (i = 0; i < 3; i++)
            {
                char temp[150];

                sprintf(temp,
                    "Train ID: %d | %s | %s to %s | Fare: Rs.%d\n",
                    t[i].id, t[i].name, t[i].from,
                    t[i].to, t[i].fare);

                strcat(response, temp);
            }

            send_msg(sock, response);
        }

        else if (choice == 2)
        {
            sscanf(buffer, "%d %d", &choice, &id);

            found = 0;

            for (i = 0; i < 3; i++)
            {
                if (t[i].id == id)
                {
                    sprintf(response,
                        "Available Seats in Train %d: %d",
                        id, t[i].seats);

                    found = 1;
                    break;
                }
            }

            if (!found)
                strcpy(response, "Invalid Train ID");

            send_msg(sock, response);
        }

        else if (choice == 3)
        {
            char name[30];

            sscanf(buffer, "%d %d %29s %d",
                   &choice, &id, name, &seats);

            found = 0;

            for (i = 0; i < 3; i++)
            {
                if (t[i].id == id)
                {
                    found = 1;
                    break;
                }
            }

            if (!found || seats <= 0 || count >= 100)
            {
                strcpy(response, "Invalid booking request");
            }
            else if (t[i].seats < seats)
            {
                strcpy(response, "Not enough seats available");
            }
            else
            {
                t[i].seats -= seats;

                b[count].id = booking_id;
                b[count].train = id;
                b[count].seats = seats;
                strcpy(b[count].name, name);
                b[count].active = 1;

                sprintf(response,
                    "Booking Successful! Booking ID: %d",
                    booking_id);

                char logmsg[100];

                sprintf(logmsg,
                    "Ticket booked by %s. Booking ID: %d",
                    name, booking_id);

                log_activity(logmsg);

                booking_id++;
                count++;
            }

            send_msg(sock, response);
        }

        else if (choice == 4)
        {
            sscanf(buffer, "%d %d", &choice, &id);

            found = 0;

            for (i = 0; i < count; i++)
            {
                if (b[i].id == id && b[i].active)
                {
                    for (int j = 0; j < 3; j++)
                    {
                        if (t[j].id == b[i].train)
                        {
                            t[j].seats += b[i].seats;
                            break;
                        }
                    }

                    b[i].active = 0;

                    strcpy(response, "Reservation Cancelled Successfully");

                    log_activity("Ticket cancellation completed");

                    found = 1;
                    break;
                }
            }

            if (!found)
                strcpy(response, "Invalid Booking ID");

            send_msg(sock, response);
        }

        else if (choice == 5)
        {
            sscanf(buffer, "%d %d", &choice, &id);

            found = 0;

            for (i = 0; i < count; i++)
            {
                if (b[i].id == id)
                {
                    if (b[i].active)
                        sprintf(response,
                            "Booking ID: %d | Name: %s | Train: %d | Seats: %d | CONFIRMED",
                            b[i].id, b[i].name,
                            b[i].train, b[i].seats);
                    else
                        strcpy(response, "Booking Cancelled");

                    found = 1;
                    break;
                }
            }

            if (!found)
                strcpy(response, "Booking Not Found");

            send_msg(sock, response);
        }

        else if (choice == 6)
        {
            send_msg(sock, "THANK YOU FOR USING RAILWAY RESERVATION");
            break;
        }

        else
        {
            send_msg(sock, "Invalid Choice");
        }
    }

    close(sock);
    printf("Client disconnected.\n");
}

int main()
{
    int server, client;
    struct sockaddr_in addr;
    int opt = 1;

    server = socket(AF_INET, SOCK_STREAM, 0);

    setsockopt(server, SOL_SOCKET, SO_REUSEADDR,
               &opt, sizeof(opt));

    addr.sin_family = AF_INET;
    addr.sin_port = htons(PORT);
    addr.sin_addr.s_addr = INADDR_ANY;

    bind(server, (struct sockaddr *)&addr, sizeof(addr));

    listen(server, 5);

    printf("Railway Reservation Server Started...\n");

    while (1)
    {
        client = accept(server, NULL, NULL);

        printf("Client Connected...\n");

        handle_client(client);
    }

    close(server);
    return 0;
}

struct Appointment a[100];

int count = 0;

    FILE *fp = fopen("hospital_log.txt", "a");

void handle_client(int sock)
{
    char buffer[MAX], response[2048];
    int choice, id, i, found;

    send_msg(sock, "HOSPITAL APPOINTMENT MANAGEMENT SYSTEM");

    while (1)
    {
        receive_msg(sock, buffer);

        if (buffer[0] == '\0')
            break;

        sscanf(buffer, "%d", &choice);

        if (choice == 1)
        {
            strcpy(response, "AVAILABLE DOCTORS:\n");

            for (i = 0; i < 3; i++)
            {
                char temp[150];

                sprintf(temp,
                    "Doctor ID: %d | Dr.%s | %s | Available Slots: %d\n",
                    d[i].id, d[i].name,
                    d[i].specialization, d[i].slots);

                strcat(response, temp);
            }

            send_msg(sock, response);
        }

        else if (choice == 2)
        {
            sscanf(buffer, "%d %d", &choice, &id);

            found = 0;

            for (i = 0; i < 3; i++)
            {
                if (d[i].id == id)
                {
                    sprintf(response,
                        "Dr.%s - Available Appointment Slots: %d",
                        d[i].name, d[i].slots);

                    found = 1;
                    break;
                }
            }

            if (!found)
                strcpy(response, "Invalid Doctor ID");

            send_msg(sock, response);
        }

        else if (choice == 3)
        {
            char name[30];

            sscanf(buffer, "%d %d %29s",
                   &choice, &id, name);

            found = 0;

            for (i = 0; i < 3; i++)
            {
                if (d[i].id == id)
                {
                    found = 1;
                    break;
                }
            }

            if (!found || count >= 100)
            {
                strcpy(response, "Invalid Doctor ID");
            }
            else if (d[i].slots <= 0)
            {
                strcpy(response, "No Appointment Slots Available");
            }
            else
            {
                d[i].slots--;

                a[count].id = appointment_id;
                a[count].doctor = id;
                strcpy(a[count].patient, name);
                a[count].active = 1;

                sprintf(response,
                    "Appointment Booked Successfully! Appointment ID: %d",
                    appointment_id);

                char logmsg[100];

                sprintf(logmsg,
                    "Appointment booked by %s. ID: %d",
                    name, appointment_id);

                log_activity(logmsg);

                appointment_id++;
                count++;
            }

            send_msg(sock, response);
        }

        else if (choice == 4)
        {
            sscanf(buffer, "%d %d", &choice, &id);

            found = 0;

            for (i = 0; i < count; i++)
            {
                if (a[i].id == id && a[i].active)
                {
                    for (int j = 0; j < 3; j++)
                    {
                        if (d[j].id == a[i].doctor)
                        {
                            d[j].slots++;
                            break;
                        }
                    }

                    a[i].active = 0;

                    strcpy(response,
                        "Appointment Cancelled Successfully");

                    log_activity("Appointment cancelled");

                    found = 1;
                    break;
                }
            }

            if (!found)
                strcpy(response, "Invalid Appointment ID");

            send_msg(sock, response);
        }

        else if (choice == 5)
        {
            char name[30];

            sscanf(buffer, "%d %29s", &choice, name);

            found = 0;
            strcpy(response, "APPOINTMENT HISTORY:\n");

            for (i = 0; i < count; i++)
            {
                if (strcmp(a[i].patient, name) == 0)
                {
                    char temp[150];

                    sprintf(temp,
                        "Appointment ID: %d | Doctor ID: %d | Status: %s\n",
                        a[i].id, a[i].doctor,
                        a[i].active ? "CONFIRMED" : "CANCELLED");

                    strcat(response, temp);
                    found = 1;
                }
            }

            if (!found)
                strcpy(response, "No Appointment History Found");

            send_msg(sock, response);
        }

        else if (choice == 6)
        {
            send_msg(sock, "THANK YOU FOR VISITING");

            break;
        }

        else
        {
            send_msg(sock, "Invalid Choice");
        }
    }

    close(sock);

    printf("Client disconnected.\n");
}
