1. TCP Congestion Control using NS-3



#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
#include "ns3/flow-monitor-module.h"

using namespace ns3;

NS_LOG_COMPONENT_DEFINE("TCPCongestion");

static void CwndChange(uint32_t oldCwnd, uint32_t newCwnd)
{
    std::cout << Simulator::Now().GetSeconds()
              << "\t" << newCwnd << std::endl;
}

int main(int argc, char *argv[])
{
    NodeContainer nodes;
    nodes.Create(3);

    PointToPointHelper p1, p2;

    p1.SetDeviceAttribute("DataRate", StringValue("10Mbps"));
    p1.SetChannelAttribute("Delay", StringValue("2ms"));

    p2.SetDeviceAttribute("DataRate", StringValue("2Mbps"));
    p2.SetChannelAttribute("Delay", StringValue("10ms"));

    p1.SetQueue("ns3::DropTailQueue",
                "MaxSize", StringValue("20p"));

    p2.SetQueue("ns3::DropTailQueue",
                "MaxSize", StringValue("20p"));

    NetDeviceContainer d1 = p1.Install(nodes.Get(0), nodes.Get(1));
    NetDeviceContainer d2 = p2.Install(nodes.Get(1), nodes.Get(2));

    InternetStackHelper stack;
    stack.Install(nodes);

    Ipv4AddressHelper address;

    address.SetBase("10.1.1.0", "255.255.255.0");
    Ipv4InterfaceContainer i1 = address.Assign(d1);

    address.SetBase("10.1.2.0", "255.255.255.0");
    Ipv4InterfaceContainer i2 = address.Assign(d2);

    Ipv4GlobalRoutingHelper::PopulateRoutingTables();

    uint16_t port = 8080;

    Address sinkAddress(InetSocketAddress(
        Ipv4Address::GetAny(), port));

    PacketSinkHelper sink("ns3::TcpSocketFactory",
                          sinkAddress);

    ApplicationContainer sinkApp = sink.Install(nodes.Get(2));

    sinkApp.Start(Seconds(0.0));
    sinkApp.Stop(Seconds(10.0));

    BulkSendHelper source("ns3::TcpSocketFactory",
        InetSocketAddress(i2.GetAddress(1), port));

    source.SetAttribute("MaxBytes", UintegerValue(0));

    ApplicationContainer sourceApp = source.Install(nodes.Get(0));

    sourceApp.Start(Seconds(1.0));
    sourceApp.Stop(Seconds(9.0));

    Simulator::Schedule(Seconds(1.1), []()
    {
        Config::ConnectWithoutContext(
        "/NodeList/0/$ns3::TcpL4Protocol/SocketList/0/CongestionWindow",
        MakeCallback(&CwndChange));
    });

    FlowMonitorHelper flowHelper;
    Ptr<FlowMonitor> monitor = flowHelper.InstallAll();

    Simulator::Stop(Seconds(10.0));

    Simulator::Run();

    monitor->CheckForLostPackets();

    Ptr<Ipv4FlowClassifier> classifier =
        DynamicCast<Ipv4FlowClassifier>(
            flowHelper.GetClassifier());

    auto stats = monitor->GetFlowStats();

    for (auto it = stats.begin(); it != stats.end(); it++)
    {
        Ipv4FlowClassifier::FiveTuple tuple =
            classifier->FindFlow(it->first);

        if (tuple.destinationPort == port)
        {
            double throughput =
                it->second.rxBytes * 8.0 / 8.0 / 1000000;

            double delay = 0;

            if (it->second.rxPackets > 0)
                delay = it->second.delaySum.GetSeconds()
                        / it->second.rxPackets;

            std::cout << "\nThroughput: "
                      << throughput << " Mbps";

            std::cout << "\nReceived Packets: "
                      << it->second.rxPackets;

            std::cout << "\nLost Packets: "
                      << it->second.lostPackets;

            std::cout << "\nAverage Delay: "
                      << delay << " seconds\n";
        }
    }

    Simulator::Destroy();

    return 0;
}


2. Comparison of TCP Reno and TCP Cubic



#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
#include "ns3/flow-monitor-module.h"

using namespace ns3;

int main(int argc, char *argv[])
{
    std::string tcp = "ns3::TcpNewReno";

    CommandLine cmd;
    cmd.AddValue("tcp", "TCP Algorithm", tcp);
    cmd.Parse(argc, argv);

    TypeId tcpType = TypeId::LookupByName(tcp);

    Config::SetDefault(
        "ns3::TcpL4Protocol::SocketType",
        TypeIdValue(tcpType));

    NodeContainer nodes;
    nodes.Create(4);

    PointToPointHelper access, bottleneck;

    access.SetDeviceAttribute(
        "DataRate", StringValue("10Mbps"));

    access.SetChannelAttribute(
        "Delay", StringValue("2ms"));

    bottleneck.SetDeviceAttribute(
        "DataRate", StringValue("2Mbps"));

    bottleneck.SetChannelAttribute(
        "Delay", StringValue("20ms"));

    bottleneck.SetQueue(
        "ns3::DropTailQueue",
        "MaxSize", StringValue("10p"));

    NetDeviceContainer d1 =
        access.Install(nodes.Get(0), nodes.Get(2));

    NetDeviceContainer d2 =
        access.Install(nodes.Get(1), nodes.Get(2));

    NetDeviceContainer d3 =
        bottleneck.Install(nodes.Get(2), nodes.Get(3));

    InternetStackHelper stack;
    stack.Install(nodes);

    Ipv4AddressHelper address;

    address.SetBase("10.1.1.0", "255.255.255.0");
    address.Assign(d1);

    address.SetBase("10.1.2.0", "255.255.255.0");
    address.Assign(d2);

    address.SetBase("10.1.3.0", "255.255.255.0");
    Ipv4InterfaceContainer i3 = address.Assign(d3);

    Ipv4GlobalRoutingHelper::PopulateRoutingTables();

    uint16_t port1 = 8080;
    uint16_t port2 = 8081;

    PacketSinkHelper sink1(
        "ns3::TcpSocketFactory",
        InetSocketAddress(Ipv4Address::GetAny(), port1));

    PacketSinkHelper sink2(
        "ns3::TcpSocketFactory",
        InetSocketAddress(Ipv4Address::GetAny(), port2));

    ApplicationContainer s1 = sink1.Install(nodes.Get(3));
    ApplicationContainer s2 = sink2.Install(nodes.Get(3));

    s1.Start(Seconds(0));
    s2.Start(Seconds(0));

    s1.Stop(Seconds(20));
    s2.Stop(Seconds(20));

    BulkSendHelper sender1(
        "ns3::TcpSocketFactory",
        InetSocketAddress(i3.GetAddress(1), port1));

    BulkSendHelper sender2(
        "ns3::TcpSocketFactory",
        InetSocketAddress(i3.GetAddress(1), port2));

    sender1.SetAttribute("MaxBytes", UintegerValue(0));
    sender2.SetAttribute("MaxBytes", UintegerValue(0));

    ApplicationContainer a1 = sender1.Install(nodes.Get(0));
    ApplicationContainer a2 = sender2.Install(nodes.Get(1));

    a1.Start(Seconds(1));
    a2.Start(Seconds(1));

    a1.Stop(Seconds(19));
    a2.Stop(Seconds(19));

    FlowMonitorHelper helper;
    Ptr<FlowMonitor> monitor = helper.InstallAll();

    Simulator::Stop(Seconds(20));
    Simulator::Run();

    monitor->CheckForLostPackets();

    Ptr<Ipv4FlowClassifier> classifier =
        DynamicCast<Ipv4FlowClassifier>(
            helper.GetClassifier());

    auto stats = monitor->GetFlowStats();

    double throughput[2] = {0, 0};
    double delay[2] = {0, 0};
    double loss[2] = {0, 0};

    int index = 0;

    for (auto it = stats.begin(); it != stats.end(); it++)
    {
        auto tuple = classifier->FindFlow(it->first);

        if (tuple.destinationPort == port1 ||
            tuple.destinationPort == port2)
        {
            index = (tuple.destinationPort == port1) ? 0 : 1;

            throughput[index] =
                it->second.rxBytes * 8.0 /
                18.0 / 1000000;

            if (it->second.rxPackets > 0)
                delay[index] =
                    it->second.delaySum.GetSeconds() /
                    it->second.rxPackets;

            loss[index] = it->second.lostPackets;
        }
    }

    double sum = throughput[0] + throughput[1];

    double fairness = 0;

    if (throughput[0] + throughput[1] > 0)
    {
        fairness = (sum * sum) /
            (2 * (throughput[0] * throughput[0] +
                  throughput[1] * throughput[1]));
    }

    std::cout << "\nTCP Algorithm: " << tcp << "\n";

    std::cout << "\nFlow 1 Throughput: "
              << throughput[0] << " Mbps";

    std::cout << "\nFlow 2 Throughput: "
              << throughput[1] << " Mbps";

    std::cout << "\nFlow 1 Delay: "
              << delay[0] << " seconds";

    std::cout << "\nFlow 2 Delay: "
              << delay[1] << " seconds";

    std::cout << "\nTotal Packet Loss: "
              << loss[0] + loss[1];

    std::cout << "\nFairness Index: "
              << fairness << "\n";

    Simulator::Destroy();

    return 0;
}
