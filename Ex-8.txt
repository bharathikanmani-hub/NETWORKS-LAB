LeetCode 239 – Sliding Window Maximum


#include <stdio.h>

int main()
{
    int n, k, i, j, max;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int a[n];

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter window size: ");
    scanf("%d", &k);

    if(k <= 0 || k > n)
    {
        printf("Invalid window size");
        return 0;
    }

    printf("Sliding Window Maximum: ");

    for(i = 0; i <= n-k; i++)
    {
        max = a[i];

        for(j = i; j < i+k; j++)
        {
            if(a[j] > max)
                max = a[j];
        }

        printf("%d ", max);
    }

    return 0;
}

2. LeetCode 480 – Sliding Window Median

#include <stdio.h>

void sort(int a[], int n)
{
    int i, j, temp;

    for(i = 0; i < n-1; i++)
    {
        for(j = 0; j < n-i-1; j++)
        {
            if(a[j] > a[j+1])
            {
                temp = a[j];
                a[j] = a[j+1];
                a[j+1] = temp;
            }
        }
    }
}

int main()
{
    int n, k, i, j;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int a[n];

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter window size: ");
    scanf("%d", &k);

    if(k <= 0 || k > n)
    {
        printf("Invalid window size");
        return 0;
    }

    int b[k];

    printf("Sliding Window Median:\n");

    for(i = 0; i <= n-k; i++)
    {
        for(j = 0; j < k; j++)
        {
            b[j] = a[i+j];
        }

        sort(b, k);

        if(k % 2 == 1)
        {
            printf("%.1f ", (double)b[k/2]);
        }
        else
        {
            printf("%.1f ",
                ((double)b[k/2-1] + b[k/2]) / 2.0);
        }
    }

    return 0;
}
